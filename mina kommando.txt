

# Ta bort  node_modules & package-lock.json (eller yarn.lock)
rd /s /q node_modules
del package-lock.json
npm install

# Ta bort Android builds
rd /s /q android\build
rd /s /q android\app\build
rd /s /q android\.cxx
rd /s /q android\app\.cxx

# Ta bort ev. genererade mappar för autolinking
rd /s /q android\app\build\generated
npm install


cd android
gradlew clean
cd ..


Rensa React Native cache
npx react-native-clean-project --remove-builds --remove-node-modules

npx react-native run-android

// ELLER Expo client
npx expo run:android


// bygga om appen
eas build --platform android --profile development


//Start appen:
npx expo --dev-client
